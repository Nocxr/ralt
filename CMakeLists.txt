cmake_minimum_required(VERSION 3.24)
project(rAlt VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "rAlt supports Windows only")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(RALT_BUILD_STANDALONE "Build the standalone tray app" ${PROJECT_IS_TOP_LEVEL})
add_library(RAltCore STATIC src/RAltEngine.cpp)
add_library(rAlt::Core ALIAS RAltCore)
target_compile_features(RAltCore PUBLIC cxx_std_20)
target_include_directories(RAltCore PUBLIC include)
target_compile_definitions(RAltCore PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(RAltCore PUBLIC user32 gdi32 shell32)
if(MSVC)
    target_compile_options(RAltCore PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(RAltCore PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(RALT_BUILD_STANDALONE)
    add_executable(rAlt WIN32 src/main.cpp)
    target_link_libraries(rAlt PRIVATE rAlt::Core)
    if(MINGW)
        target_link_options(rAlt PRIVATE -municode)
    endif()
endif()
